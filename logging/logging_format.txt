

                    _______________________logging.basicConfig()________________________________

logger = logging.getLogger(__name__)

logger.setLevel(logging.DEBUG)

console_handler = logging.StreamHandler()
console_handler.setLevel(level=logging.DEBUG)
con_format = logging.Formatter("%(levelname)s: %(message)s")
console_handler.setFormatter(con_format)

file_handler = logging.FileHandler(filename='app.log', mode='w')
file_handler.setLevel(logging.WARNING)
file_format = logging.Formatter("%(asctime)s - %(name)s at line %(lineno)d - %(levelname)s: %(message)s")
file_format = logging.Formatter("""
                                time       - %(asctime)s 
                                filename   - %(filename)s %(module)s
                                funcname   - %(funcName)s
                                at line    - %(lineno)d 
                                loglevel   - %(levelname)s %(levelno)d
                                logmessage - %(message)s
                                pathname - %(pathname)s
                                """)
file_handler.setFormatter(file_format)


logger.addHandler(console_handler)
logger.addHandler(file_handler)
